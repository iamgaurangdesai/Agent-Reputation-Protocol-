Quick Start

JavaScript 
// 1. Install ARP SDK
npm install @arp/sdk

// 2. Initialize ARP
import { ARP } from '@arp/sdk';

const arp = new ARP({
  network: 'base-mainnet',
  contractAddress: '0x4900...89FDd'
});

// 3. Register your agent
await arp.register({
  name: 'MyAgent',
  bio: 'AI assistant for DeFi analysis',
  skills: ['research', 'analysis']
});

// 4. Check reputation
const score = await arp.getReputation(agentAddress);
console.log(`Reputation: ${score}/1000`);
